### 7.1.1 个人博客系统

#### 项目概述

```sql
CREATE TABLE `user` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '用户ID',
    `username` VARCHAR(50) NOT NULL COMMENT '用户名',
    `password` VARCHAR(100) NOT NULL COMMENT '密码（BCrypt加密）',
    `nickname` VARCHAR(50) DEFAULT NULL COMMENT '昵称',
    `email` VARCHAR(100) DEFAULT NULL COMMENT '邮箱',
    `avatar` VARCHAR(255) DEFAULT NULL COMMENT '头像URL',
    `role` TINYINT NOT NULL DEFAULT 0 COMMENT '角色：0-普通用户，1-管理员',
    `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态：0-禁用，1-启用',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_username` (`username`),
    UNIQUE KEY `uk_email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 分类表
CREATE TABLE `category` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '分类ID',
    `name` VARCHAR(50) NOT NULL COMMENT '分类名称',
    `description` VARCHAR(200) DEFAULT NULL COMMENT '分类描述',
    `sort_order` INT NOT NULL DEFAULT 0 COMMENT '排序序号',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='分类表';

-- 标签表
CREATE TABLE `tag` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '标签ID',
    `name` VARCHAR(50) NOT NULL COMMENT '标签名称',
    `color` VARCHAR(20) DEFAULT '#409EFF' COMMENT '标签颜色',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='标签表';

-- 文章表
CREATE TABLE `article` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '文章ID',
    `title` VARCHAR(200) NOT NULL COMMENT '文章标题',
    `summary` VARCHAR(500) DEFAULT NULL COMMENT '文章摘要',
    `content` LONGTEXT NOT NULL COMMENT '文章内容（Markdown）',
    `cover_image` VARCHAR(255) DEFAULT NULL COMMENT '封面图片URL',
    `author_id` BIGINT NOT NULL COMMENT '作者ID',
    `category_id` BIGINT DEFAULT NULL COMMENT '分类ID',
    `status` TINYINT NOT NULL DEFAULT 0 COMMENT '状态：0-草稿，1-已发布',
    `view_count` INT NOT NULL DEFAULT 0 COMMENT '浏览次数',
    `like_count` INT NOT NULL DEFAULT 0 COMMENT '点赞数',
    `comment_count` INT NOT NULL DEFAULT 0 COMMENT '评论数',
    `is_top` TINYINT NOT NULL DEFAULT 0 COMMENT '是否置顶：0-否，1-是',
    `is_deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '逻辑删除：0-未删除，1-已删除',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    KEY `idx_author_id` (`author_id`),
    KEY `idx_category_id` (`category_id`),
    KEY `idx_status` (`status`),
    KEY `idx_created_at` (`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='文章表';

-- 文章标签关联表
CREATE TABLE `article_tag` (
    `article_id` BIGINT NOT NULL COMMENT '文章ID',

    `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态：0-待审核，1-已通过',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    KEY `idx_article_id` (`article_id`),
    KEY `idx_user_id` (`user_id`),
    KEY `idx_parent_id` (`parent_id`)
```

```java
    /**
     * 业务异常
     */
    @ExceptionHandler(BusinessException.class)
    public Result<Void> handleBusinessException(BusinessException e) {
        log.warn("业务异常：{}", e.getMessage());
        return Result.error(e.getCode(), e.getMessage());
    }

    /**
     * 参数校验异常
     */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Void> handleValidException(MethodArgumentNotValidException e) {
        String message = e.getBindingResult().getAllErrors().get(0).getDefaultMessage();
        log.warn("参数校验失败：{}", message);
        return Result.error(400, message);
    }

    /**
     * 系统异常
     */
    @ExceptionHandler(Exception.class)
    public Result<Void> handleException(Exception e) {
        log.error("系统异常：", e);
```

**文章服务层 ArticleServiceImpl.java：**

```java
@Service
@Slf4j
public class ArticleServiceImpl extends ServiceImpl<ArticleMapper, Article>
        implements ArticleService {

    @Autowired
    private ArticleTagMapper articleTagMapper;
    @Autowired
    private CategoryMapper categoryMapper;
    @Autowired
    private UserMapper userMapper;
    @Autowired
    private StringRedisTemplate redisTemplate;

    private static final String ARTICLE_CACHE_KEY = "blog:article:";
    private static final String ARTICLE_LIST_CACHE_KEY = "blog:article:list:";

    /**
     * 发布文章
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public Long publishArticle(ArticleDTO dto) {
        // 1. 构建文章实体
        Article article = new Article();
        BeanUtils.copyProperties(dto, article);
        article.setStatus(1); // 已发布

        // 2. 保存文章
        this.save(article);

        // 3. 保存文章标签关联
        if (dto.getTagIds() != null && !dto.getTagIds().isEmpty()) {
            for (Long tagId : dto.getTagIds()) {
                ArticleTag articleTag = new ArticleTag(article.getId(), tagId);
                articleTagMapper.insert(articleTag);
            }
        }

        // 4. 清除缓存
        redisTemplate.delete(ARTICLE_LIST_CACHE_KEY + "*");

        log.info("文章发布成功，ID：{}", article.getId());
        return article.getId();
    }

    /**
     * 分页查询文章列表（带缓存）
     */
    @Override
    public PageResult<ArticleVO> getArticleList(ArticleQueryDTO query) {
        String cacheKey = ARTICLE_LIST_CACHE_KEY + query.getPage() + ":" + query.getSize();

        // 尝试从缓存获取
        String cached = redisTemplate.opsForValue().get(cacheKey);
        if (cached != null) {
            try {
                return new ObjectMapper().readValue(cached,
                    new TypeReference<PageResult<ArticleVO>>() {});
            } catch (JsonProcessingException e) {
                log.warn("缓存反序列化失败，重新查询");
            }
        }

        // 构建查询条件
        Page<Article> page = new Page<>(query.getPage(), query.getSize());
        LambdaQueryWrapper<Article> wrapper = new LambdaQueryWrapper<>();

        wrapper.eq(Article::getStatus, 1)
               .eq(query.getCategoryId() != null, Article::getCategoryId, query.getCategoryId())
               .like(StringUtils.hasText(query.getKeyword()), Article::getTitle, query.getKeyword())
               .orderByDesc(Article::getIsTop)
               .orderByDesc(Article::getCreatedAt);

        Page<Article> result = this.page(page, wrapper);

        // 转换为VO
        List<ArticleVO> voList = result.getRecords().stream()
                .map(this::convertToVO)
                .collect(Collectors.toList());

        PageResult<ArticleVO> pageResult = new PageResult<>(
                voList, result.getTotal(), query.getPage(), query.getSize());

        // 写入缓存（30分钟过期）
        try {
            redisTemplate.opsForValue().set(cacheKey,
                new ObjectMapper().writeValueAsString(pageResult),
                30, TimeUnit.MINUTES);
        } catch (JsonProcessingException e) {
            log.warn("缓存序列化失败");
        }

        return pageResult;
    }

    /**
     * 获取文章详情
     */
    @Override
    public ArticleDetailVO getArticleDetail(Long id) {
        // 尝试从缓存获取
        String cacheKey = ARTICLE_CACHE_KEY + id;
        String cached = redisTemplate.opsForValue().get(cacheKey);
        if (cached != null) {
            try {
                return new ObjectMapper().readValue(cached, ArticleDetailVO.class);
            } catch (JsonProcessingException e) {
                log.warn("缓存反序列化失败");
            }
        }

        // 查询文章
        Article article = this.getById(id);
        if (article == null || article.getStatus() != 1) {
            throw new BusinessException(404, "文章不存在");
        }

        // 增加浏览次数
        this.update(new UpdateWrapper<Article>()
                .eq("id", id)
                .setSql("view_count = view_count + 1"));

        // 转换为详情VO
        ArticleDetailVO vo = new ArticleDetailVO();
        BeanUtils.copyProperties(article, vo);
        vo.setViewCount(article.getViewCount() + 1);

        // 查询作者信息
        User author = userMapper.selectById(article.getAuthorId());
        vo.setAuthorName(author.getNickname());
        vo.setAuthorAvatar(author.getAvatar());

        // 查询分类信息
        if (article.getCategoryId() != null) {
            Category category = categoryMapper.selectById(article.getCategoryId());
            vo.setCategoryName(category.getName());
        }

        // 查询标签列表
        List<Tag> tags = articleTagMapper.selectTagsByArticleId(id);
        vo.setTags(tags);

        // 写入缓存
        try {
            redisTemplate.opsForValue().set(cacheKey,
                new ObjectMapper().writeValueAsString(vo),
                1, TimeUnit.HOURS);
        } catch (JsonProcessingException e) {
            log.warn("缓存序列化失败");
        }

        return vo;
    }

    /**
     * 转换为VO
     */
    private ArticleVO convertToVO(Article article) {
        ArticleVO vo = new ArticleVO();
        BeanUtils.copyProperties(article, vo);

        // 查询作者
```

**JWT工具类 JwtUtil.java：**

```java
@Component
public class JwtUtil {

    @Value("${jwt.secret}")
    private String secret;

    @Value("${jwt.expiration}")
    private long expiration;

    /**
     * 生成Token
     */
    public String generateToken(Long userId, String username) {
        return Jwts.builder()
                .subject(username)
                .claim("userId", userId)
                .issuedAt(new Date())
                .expiration(new Date(System.currentTimeMillis() + expiration))
                .signWith(getSigningKey())
                .compact();
    }

    /**
     * 解析Token
     */
    public Claims parseToken(String token) {
        return Jwts.parser()
                .setSigningKey(getSigningKey())
                .build()
                .parseClaimsJws(token)
                .getBody();
    }

    /**
     * 验证Token
     */
    public boolean validateToken(String token) {
        try {
            parseToken(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    /**
     * 获取用户ID
     */
    public Long getUserId(String token) {
        Claims claims = parseToken(token);
        return claims.get("userId", Long.class);
    }

    private SecretKey getSigningKey() {
        byte[] keyBytes = Decoders.BASE64.decode(
            Base64.getEncoder().encodeToString(secret.getBytes()));
        return Keys.hmacShaKeyFor(keyBytes);
    }
}
```

```java
@Component
public class JwtInterceptor implements HandlerInterceptor {

    @Autowired
    private JwtUtil jwtUtil;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response,
                             Object handler) {
        // 放行OPTIONS预检请求
        if ("OPTIONS".equalsIgnoreCase(request.getMethod())) {
            return true;
        }

        String token = request.getHeader("Authorization");
        if (StringUtils.hasText(token) && token.startsWith("Bearer ")) {
            token = token.substring(7);
        }

        if (!StringUtils.hasText(token) || !jwtUtil.validateToken(token)) {
            response.setStatus(401);
            response.setContentType("application/json;charset=UTF-8");
            try {
                response.getWriter().write("{\"code\":401,\"message\":\"未登录或Token已过期\"}");
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
            return false;
        }

        // 将用户信息存入request属性
        Long userId = jwtUtil.getUserId(token);
        request.setAttribute("userId", userId);
        return true;
    }
}
```

**雪花算法ID生成器 SnowflakeIdGenerator.java：**

```java
public class SnowflakeIdGenerator {

    private final long epoch = 1704067200000L; // 2024-01-01 00:00:00
    private final long workerIdBits = 5L;
    private final long datacenterIdBits = 5L;
    private final long sequenceBits = 12L;

    private final long maxWorkerId = ~(-1L << workerIdBits);
    private final long maxDatacenterId = ~(-1L << datacenterIdBits);
    private final long sequenceMask = ~(-1L << sequenceBits);

    private final long workerIdShift = sequenceBits;
    private final long datacenterIdShift = sequenceBits + workerIdBits;
    private final long timestampShift = sequenceBits + workerIdBits + datacenterIdBits;

    private long workerId;
    private long datacenterId;
    private long sequence = 0L;
    private long lastTimestamp = -1L;

    public SnowflakeIdGenerator(long workerId, long datacenterId) {
        if (workerId > maxWorkerId || workerId < 0) {
            throw new IllegalArgumentException("workerId不合法");
        }
        if (datacenterId > maxDatacenterId || datacenterId < 0) {
            throw new IllegalArgumentException("datacenterId不合法");
        }
        this.workerId = workerId;
        this.datacenterId = datacenterId;
    }

    public synchronized long nextId() {
        long timestamp = System.currentTimeMillis();

        if (timestamp < lastTimestamp) {
            throw new RuntimeException("时钟回拨，拒绝生成ID");
        }

        if (lastTimestamp == timestamp) {
            sequence = (sequence + 1) & sequenceMask;
            if (sequence == 0) {
                timestamp = tilNextMillis(lastTimestamp);
            }
        } else {
            sequence = 0L;
        }

        lastTimestamp = timestamp;

        return ((timestamp - epoch) << timestampShift)
                | (datacenterId << datacenterIdShift)
                | (workerId << workerIdShift)
                | sequence;
    }

    private long tilNextMillis(long lastTimestamp) {
        long timestamp = System.currentTimeMillis();
        while (timestamp <= lastTimestamp) {
            timestamp = System.currentTimeMillis();
        }
        return timestamp;
    }
}
```

```java
@Slf4j
public class OrderServiceImpl extends ServiceImpl<OrderMapper, Order>
        implements OrderService {

    @Autowired

    @Autowired

    @Autowired

    /**
     * 创建订单（核心流程）
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public String createOrder(CreateOrderDTO dto, Long userId) {
        log.info("开始创建订单，用户ID：{}", userId);

        // 1. 获取收货地址
        UserAddressDTO address = userFeignClient.getAddress(dto.getAddressId());
        if (address == null) {
            throw new BusinessException("收货地址不存在");
        }

        // 2. 获取商品信息并计算总价
        BigDecimal totalAmount = BigDecimal.ZERO;
        List<OrderItem> orderItems = new ArrayList<>();

        for (OrderItemDTO itemDTO : dto.getItems()) {
            // 查询SKU信息
            SkuDTO sku = productFeignClient.getSku(itemDTO.getSkuId());
            if (sku == null) {
                throw new BusinessException("商品SKU不存在：" + itemDTO.getSkuId());
            }

            // 检查库存
            if (sku.getStock() - sku.getLockedStock() < itemDTO.getQuantity()) {
                throw new BusinessException("商品库存不足：" + sku.getSkuName());
            }

            // 构建订单明细
            OrderItem item = new OrderItem();
            item.setProductId(sku.getProductId());
            item.setSkuId(sku.getId());
            item.setProductName(sku.getProductName());
            item.setSkuName(sku.getSkuName());
            item.setProductImage(sku.getImage());
            item.setPrice(sku.getPrice());
            item.setQuantity(itemDTO.getQuantity());
            item.setTotalAmount(sku.getPrice().multiply(
                new BigDecimal(itemDTO.getQuantity())));
            orderItems.add(item);

            totalAmount = totalAmount.add(item.getTotalAmount());
        }

        // 3. 生成订单编号
        String orderNo = generateOrderNo(userId);

        // 4. 创建订单
        Order order = new Order();
        order.setId(idGenerator.nextId());
        order.setOrderNo(orderNo);
        order.setUserId(userId);
        order.setTotalAmount(totalAmount);
        order.setPayAmount(totalAmount); // 暂不计算优惠
        order.setFreightAmount(BigDecimal.ZERO);
        order.setDiscountAmount(BigDecimal.ZERO);
        order.setStatus(OrderConstants.STATUS_PENDING);
        order.setReceiverName(address.getReceiverName());
        order.setReceiverPhone(address.getReceiverPhone());
        order.setReceiverAddress(
            address.getProvince() + address.getCity() +
            address.getDistrict() + address.getDetailAddress());
        order.setExpireTime(LocalDateTime.now().plusMinutes(30)); // 30分钟过期

        this.save(order);

        // 5. 保存订单明细
        for (OrderItem item : orderItems) {
            item.setOrderId(order.getId());
            item.setOrderNo(orderNo);
            orderItemMapper.insert(item);
        }

        // 6. 锁定库存（调用商品服务）
        List<StockDeductionDTO> stockList = dto.getItems().stream()
                .map(item -> new StockDeductionDTO(item.getSkuId(), item.getQuantity()))
                .collect(Collectors.toList());
        productFeignClient.lockStock(stockList);

        // 7. 发送延迟消息（30分钟后自动取消未支付订单）
        rabbitTemplate.convertAndSend(
            "order.exchange", "order.delay",
            orderNo,
            message -> {
                message.getMessageProperties().setDelay(30 * 60 * 1000);
                return message;
            }
        );

        // 8. 清除购物车（异步）
        List<Long> skuIds = dto.getItems().stream()
                .map(OrderItemDTO::getSkuId)
                .collect(Collectors.toList());
        rabbitTemplate.convertAndSend("cart.exchange", "cart.clear",
            new CartClearMessage(userId, skuIds));

        log.info("订单创建成功，订单号：{}", orderNo);
        return orderNo;
    }

    /**
     * 支付订单

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void payOrder(String orderNo, String payChannel) {
        Order order = this.getOne(
            new LambdaQueryWrapper<Order>().eq(Order::getOrderNo, orderNo));

        if (order == null) {
            throw new BusinessException("订单不存在");
        }
        if (order.getStatus() != OrderConstants.STATUS_PENDING) {
            throw new BusinessException("订单状态异常");
        }

        // 调用支付服务（通过Feign）
        // paymentFeignClient.pay(orderNo, payChannel, order.getPayAmount());

        // 更新订单状态
        order.setStatus(OrderConstants.STATUS_PAID);
        order.setPaymentTime(LocalDateTime.now());
        this.updateById(order);

        // 扣减库存（确认）
        productFeignClient.confirmStock(orderNo);

        // 发送订单已支付消息
        rabbitTemplate.convertAndSend("order.exchange", "order.paid", orderNo);
    }

    /**
     * 取消订单
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public void cancelOrder(String orderNo) {
        Order order = this.getOne(
            new LambdaQueryWrapper<Order>().eq(Order::getOrderNo, orderNo));

        if (order == null) {
            throw new BusinessException("订单不存在");
        }
        if (order.getStatus() != OrderConstants.STATUS_PENDING) {
            throw new BusinessException("只能取消待支付订单");
        }

        // 更新订单状态
        order.setStatus(OrderConstants.STATUS_CANCELLED);
        this.updateById(order);

        // 释放库存
        productFeignClient.unlockStock(orderNo);

        log.info("订单已取消，订单号：{}", orderNo);
    }

        String timestamp = String.valueOf(System.currentTimeMillis());
        String userSuffix = String.format("%04d", userId % 10000);
        String random = String.format("%04d", new Random().nextInt(10000));
        return timestamp + userSuffix + random;
    }
```

```java
@Component
public class AuthGlobalFilter implements GlobalFilter, Ordered {

    private static final List<String> WHITE_LIST = List.of(
        "/api/user/login",
        "/api/user/register",
        "/api/product/list",
        "/api/product/detail"
    );

    @Autowired
    private JwtUtil jwtUtil;

    @Override
    public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) {
        ServerHttpRequest request = exchange.getRequest();
        String path = request.getURI().getPath();

        // 白名单直接放行
        if (WHITE_LIST.stream().anyMatch(path::startsWith)) {
            return chain.filter(exchange);
        }

        // 获取Token
        String token = request.getHeaders().getFirst("Authorization");
        if (!StringUtils.hasText(token) || !token.startsWith("Bearer ")) {
            return unauthorized(exchange);
        }

        token = token.substring(7);

        // 验证Token
        if (!jwtUtil.validateToken(token)) {
            return unauthorized(exchange);
        }

        // 将用户信息传递到下游服务
        Long userId = jwtUtil.getUserId(token);
        ServerHttpRequest mutatedRequest = request.mutate()
                .header("X-User-Id", userId.toString())
                .build();

        return chain.filter(exchange.mutate().request(mutatedRequest).build());
    }

    @Override
    public int getOrder() {
        return -1;
    }

    private Mono<Void> unauthorized(ServerWebExchange exchange) {
        ServerHttpResponse response = exchange.getResponse();
        response.setStatusCode(HttpStatus.UNAUTHORIZED);
        response.getHeaders().setContentType(MediaType.APPLICATION_JSON);
        String body = "{\"code\":401,\"message\":\"未登录或Token已过期\"}";
        DataBuffer buffer = response.bufferFactory()
                .wrap(body.getBytes(StandardCharsets.UTF_8));
        return response.writeWith(Mono.just(buffer));
    }
}
```

```java
@Component
@Slf4j
public class NettyServer {

    @Value("${netty.port:8888}")
    private int port;

    @Autowired
    private WebSocketChannelHandler channelHandler;

    private EventLoopGroup bossGroup;
    private EventLoopGroup workerGroup;

    @PostConstruct
    public void start() {
        new Thread(this::init).start();
    }

    private void init() {
        bossGroup = new NioEventLoopGroup(1);
        workerGroup = new NioEventLoopGroup();

        try {
            ServerBootstrap bootstrap = new ServerBootstrap();
            bootstrap.group(bossGroup, workerGroup)
                    .channel(NioServerSocketChannel.class)
                    .option(ChannelOption.SO_BACKLOG, 1024)
                    .option(ChannelOption.SO_KEEPALIVE, true)
                    .childOption(ChannelOption.TCP_NODELAY, true)
                    .childHandler(new ChannelInitializer<SocketChannel>() {
                        @Override
                        protected void initChannel(SocketChannel ch) {
                            ChannelPipeline pipeline = ch.pipeline();
                            // HTTP编解码器（WebSocket基于HTTP升级）
                            pipeline.addLast(new HttpServerCodec());
                            pipeline.addLast(new ChunkedWriteHandler());
                            // HTTP聚合器（WebSocket握手需要）
                            pipeline.addLast(new HttpObjectAggregator(65536));
                            // WebSocket协议处理器
                            pipeline.addLast(new WebSocketServerProtocolHandler("/ws"));
                            // 心跳检测
                            pipeline.addLast(new IdleStateHandler(60, 30, 0));
                            pipeline.addLast(new HeartbeatHandler());
                            // 消息处理
                            pipeline.addLast(channelHandler);
                        }
                    });

            ChannelFuture future = bootstrap.bind(port).sync();
            log.info("Netty WebSocket服务器启动，端口：{}", port);
            future.channel().closeFuture().sync();

        } catch (InterruptedException e) {
            log.error("Netty服务器异常", e);
            Thread.currentThread().interrupt();
        } finally {
            shutdown();
        }
    }

    @PreDestroy
    public void shutdown() {
        if (bossGroup != null) bossGroup.shutdownGracefully();
        if (workerGroup != null) workerGroup.shutdownGracefully();
        log.info("Netty服务器已关闭");
    }
```

```java
@Component
@Slf4j
public class SingleChatHandler {

    @Autowired
    private SessionManager sessionManager;
    @Autowired
    private MessageService messageService;
    @Autowired
    private OnlineService onlineService;
    @Autowired
    private ObjectMapper objectMapper;

    public void handle(ChannelHandlerContext ctx, MessageProtocol protocol) {
        Long fromUserId = protocol.getFromUserId();
        Long toUserId = protocol.getToId();

        // 1. 保存消息到数据库
        Message message = new Message();
        message.setId(Long.parseLong(protocol.getMsgId()));
        message.setFromUserId(fromUserId);
        message.setToId(toUserId);
        message.setMsgType(MessageType.SINGLE_CHAT);
        message.setContentType(protocol.getContentType());
        message.setContent(protocol.getContent());
        message.setStatus(0); // 发送中
        messageService.save(message);

        // 2. 构建推送消息
        MessageProtocol pushMsg = MessageProtocol.builder()
                .type(MessageType.SINGLE_CHAT)
                .msgId(protocol.getMsgId())
                .fromUserId(fromUserId)
                .toId(toUserId)
                .contentType(protocol.getContentType())
                .content(protocol.getContent())
                .timestamp(System.currentTimeMillis())
                .build();

        // 3. 判断接收者是否在线
        if (sessionManager.isOnline(toUserId)) {
            // 在线：直接推送
            Channel toChannel = sessionManager.getChannel(toUserId);
            sendToChannel(toChannel, pushMsg);

            // 更新消息状态为已送达
            message.setStatus(1);
            messageService.updateById(message);
        } else {
            // 离线：存入离线消息队列
            onlineService.saveOfflineMessage(toUserId, pushMsg);
        }

        // 4. 给发送者回复ACK
        MessageProtocol ackMsg = MessageProtocol.builder()
                .type(MessageType.MESSAGE_ACK)
                .msgId(protocol.getMsgId())
                .timestamp(System.currentTimeMillis())
                .build();
        sendToChannel(ctx.channel(), ackMsg);
    }

    private void sendToChannel(Channel channel, MessageProtocol protocol) {
        try {
            String json = objectMapper.writeValueAsString(protocol);
            channel.writeAndFlush(new TextWebSocketFrame(json));
        } catch (JsonProcessingException e) {
            log.error("消息序列化失败", e);
        }
    }
}
```

```java
            IdleStateEvent event = (IdleStateEvent) evt;
            if (event.state() == IdleState.READER_IDLE) {

                        .timestamp(System.currentTimeMillis())
```

```java
@Component
public class Base62Generator implements ShortCodeGenerator {

    private static final String BASE62_CHARS =
        "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";
    private static final int BASE = 62;
    private static final int CODE_LENGTH = 6; // 6位短码

    @Autowired
    private IdGenerator idGenerator;

    @Override
    public String generate() {
        long id = idGenerator.nextId();
        return encode(id);
    }

    /**
     * 将数字编码为Base62字符串
     */
    private String encode(long num) {
        StringBuilder sb = new StringBuilder();
        while (num > 0) {
            sb.insert(0, BASE62_CHARS.charAt((int) (num % BASE)));
            num /= BASE;
        }
        // 补齐到固定长度
        while (sb.length() < CODE_LENGTH) {
            sb.insert(0, '0');
        }
        return sb.toString();
    }

    /**
     * 将Base62字符串解码为数字
     */
    public long decode(String code) {
        long num = 0;
        for (int i = 0; i < code.length(); i++) {
            num = num * BASE + BASE62_CHARS.indexOf(code.charAt(i));
        }
        return num;
    }
}
```

**预生成池方案 PreGeneratedPoolGenerator.java：**

```java
@Component
public class PreGeneratedPoolGenerator implements ShortCodeGenerator {

    @Autowired
    private StringRedisTemplate redisTemplate;

    private static final String POOL_KEY = "shorturl:code:pool";
    private static final int POOL_THRESHOLD = 1000; // 低于此值触发补充

    @PostConstruct
    public void init() {
        // 初始化时预生成一批短码
        if (redisTemplate.opsForList().size(POOL_KEY) == 0) {
            fillPool(10000);
        }
        // 定时检查并补充
        ScheduledExecutorService scheduler = Executors.newSingleThreadScheduledExecutor();
        scheduler.scheduleAtFixedRate(this::checkAndFill, 1, 1, TimeUnit.MINUTES);
    }

    @Override
    public String generate() {
        String code = redisTemplate.opsForList().rightPop(POOL_KEY);
        if (code == null) {
            // 池为空，同步生成
            fillPool(1000);
            code = redisTemplate.opsForList().rightPop(POOL_KEY);
        }
        return code;
    }

    private void checkAndFill() {
        Long size = redisTemplate.opsForList().size(POOL_KEY);
        if (size != null && size < POOL_THRESHOLD) {
            fillPool(5000);
        }
    }

    private synchronized void fillPool(int count) {
        List<String> codes = new ArrayList<>();
        for (int i = 0; i < count; i++) {
            codes.add(randomCode());
        }
        redisTemplate.opsForList().leftPushAll(POOL_KEY, codes);
    }

    private String randomCode() {
        String chars = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";
        StringBuilder sb = new StringBuilder(6);
        Random random = new Random();
        for (int i = 0; i < 6; i++) {
            sb.append(chars.charAt(random.nextInt(chars.length())));
        }
        return sb.toString();
    }
}
```

**短链接服务核心 ShortUrlServiceImpl.java：**

```java
@Service
@Slf4j
public class ShortUrlServiceImpl implements ShortUrlService {

    @Autowired
    private ShortUrlMapper shortUrlMapper;
    @Autowired
    private ShortCodeGenerator shortCodeGenerator;
    @Autowired
    private StringRedisTemplate redisTemplate;
    @Autowired
    private RabbitTemplate rabbitTemplate;

    private static final String CACHE_PREFIX = "shorturl:";
    private static final long CACHE_EXPIRE_HOURS = 24;

    /**
     * 创建短链接
     */
    @Override
    public ShortUrl createShortUrl(String originalUrl, Long userId, LocalDateTime expireTime) {
        // 1. 检查是否已存在相同URL的短链接
        String existCode = redisTemplate.opsForValue().get("shorturl:url:" + originalUrl);
        if (existCode != null) {
            return shortUrlMapper.selectByShortCode(existCode);
        }

        ShortUrl exist = shortUrlMapper.selectByOriginalUrl(originalUrl);
        if (exist != null) {
            return exist;
        }

        // 2. 生成短码（保证唯一）
        String shortCode;
        do {
            shortCode = shortCodeGenerator.generate();
        } while (shortUrlMapper.selectByShortCode(shortCode) != null);

        // 3. 保存短链接
        ShortUrl shortUrl = new ShortUrl();
        shortUrl.setShortCode(shortCode);
        shortUrl.setOriginalUrl(originalUrl);
        shortUrl.setUserId(userId);
        shortUrl.setExpireTime(expireTime);
        shortUrlMapper.insert(shortUrl);

        // 4. 写入缓存
        redisTemplate.opsForValue().set(
            CACHE_PREFIX + shortCode, originalUrl,
            CACHE_EXPIRE_HOURS, TimeUnit.HOURS);
        redisTemplate.opsForValue().set(
            "shorturl:url:" + originalUrl, shortCode,
            CACHE_EXPIRE_HOURS, TimeUnit.HOURS);

        log.info("短链接创建成功：{} -> {}", shortCode, originalUrl);
        return shortUrl;
    }

    /**
     * 解析短链接（重定向核心逻辑）
     */
    @Override
    public String resolveShortUrl(String shortCode, String clientIp, String userAgent) {
        // 1. 先查缓存
        String originalUrl = redisTemplate.opsForValue().get(CACHE_PREFIX + shortCode);

        if (originalUrl == null) {
            // 2. 缓存未命中，查数据库
            ShortUrl shortUrl = shortUrlMapper.selectByShortCode(shortCode);
            if (shortUrl == null) {
                return null;
            }

            // 检查是否过期
            if (shortUrl.getExpireTime() != null &&
                shortUrl.getExpireTime().isBefore(LocalDateTime.now())) {
                return null;
            }

            // 检查是否禁用
            if (shortUrl.getStatus() != 1) {
                return null;
            }

            originalUrl = shortUrl.getOriginalUrl();

            // 3. 写入缓存
            redisTemplate.opsForValue().set(
                CACHE_PREFIX + shortCode, originalUrl,
                CACHE_EXPIRE_HOURS, TimeUnit.HOURS);
        }

        // 4. 异步记录访问日志
        UrlAccessLog accessLog = new UrlAccessLog();
        accessLog.setShortCode(shortCode);
        accessLog.setAccessIp(clientIp);
        accessLog.setUserAgent(userAgent);
        accessLog.setCreatedAt(LocalDateTime.now());
        rabbitTemplate.convertAndSend("url.access", "access.log", accessLog);

        // 5. 异步更新访问次数
        redisTemplate.opsForValue().increment("shorturl:count:" + shortCode);

        return originalUrl;
    }

    /**
     * 获取短链接统计信息
     */
    @Override
    public UrlStatistics getStatistics(String shortCode) {
        UrlStatistics stats = new UrlStatistics();

        // 总访问次数
        String countStr = redisTemplate.opsForValue().get("shorturl:count:" + shortCode);
        long redisCount = countStr != null ? Long.parseLong(countStr) : 0;

        ShortUrl shortUrl = shortUrlMapper.selectByShortCode(shortCode);
```

```java
                         HttpServletResponse response) throws IOException {

        String userAgent = request.getHeader("User-Agent");

        String originalUrl = shortUrlService.resolveShortUrl(shortCode, clientIp, userAgent);

        if (originalUrl == null) {

        response.setStatus(302);
        response.setHeader("Location", originalUrl);
        response.setHeader("Cache-Control", "no-cache");
    }

    private String getClientIp(HttpServletRequest request) {
        String ip = request.getHeader("X-Forwarded-For");
        if (ip == null || ip.isEmpty() || "unknown".equalsIgnoreCase(ip)) {
            ip = request.getHeader("X-Real-IP");
        }
        if (ip == null || ip.isEmpty() || "unknown".equalsIgnoreCase(ip)) {
            ip = request.getRemoteAddr();
        }
        // 多个代理时取第一个
        if (ip != null && ip.contains(",")) {
            ip = ip.split(",")[0].trim();
        }
        return ip;
    }
```

```java
@Component
public class RateLimitFilter implements Filter {

    @Autowired
    private StringRedisTemplate redisTemplate;

    private static final int MAX_REQUESTS = 100; // 每分钟最大请求数
    private static final int WINDOW_SECONDS = 60;

    @Override
    public void doFilter(ServletRequest request, ServletResponse response,
                         FilterChain chain) throws IOException, ServletException {
        HttpServletRequest httpRequest = (HttpServletRequest) request;
        String clientIp = httpRequest.getRemoteAddr();
        String key = "ratelimit:" + clientIp;

        Long count = redisTemplate.opsForValue().increment(key);
        if (count == 1) {
            redisTemplate.expire(key, WINDOW_SECONDS, TimeUnit.SECONDS);
        }

        if (count > MAX_REQUESTS) {
            HttpServletResponse httpResponse = (HttpServletResponse) response;
            httpResponse.setStatus(429);
            httpResponse.setContentType("application/json;charset=UTF-8");
            httpResponse.getWriter().write(
                "{\"code\":429,\"message\":\"请求过于频繁，请稍后再试\"}");
            return;
        }

        chain.doFilter(request, response);
    }
}
```

#### 项目概述

基于RBAC（基于角色的访问控制）模型的权限管理系统，支持用户管理、角色管理、菜单管理、部门管理、日志审计等功能。

#### RBAC权限模型

```
用户(User) ──N:M──> 角色(Role) ──N:M──> 权限/菜单(Permission)
     │                                    │
     └──────────── 部门(Dept) ────────────┘
```

- 一个用户可以拥有多个角色
- 一个角色可以分配给多个用户
- 一个角色可以拥有多个权限
- 一个权限可以属于多个角色

**JWT认证过滤器 JwtAuthenticationFilter.java：**

```java
@Component
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtUtil jwtUtil;
    private final UserDetailsService userDetailsService;

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                    HttpServletResponse response,
                                    FilterChain filterChain)
            throws ServletException, IOException {

        String authHeader = request.getHeader("Authorization");

        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            filterChain.doFilter(request, response);
            return;
        }

        String token = authHeader.substring(7);

        if (!jwtUtil.validateToken(token)) {
            filterChain.doFilter(request, response);
            return;
        }

        String username = jwtUtil.getUsername(token);

        // 如果尚未设置认证信息
        if (SecurityContextHolder.getContext().getAuthentication() == null) {
            UserDetails userDetails = userDetailsService.loadUserByUsername(username);

            UsernamePasswordAuthenticationToken authToken =
                new UsernamePasswordAuthenticationToken(
                    userDetails, null, userDetails.getAuthorities());

            authToken.setDetails(
                new WebAuthenticationDetailsSource().buildDetails(request));

            SecurityContextHolder.getContext().setAuthentication(authToken);
        }

        filterChain.doFilter(request, response);
    }
}
```

```java
@Service
public class UserDetailsServiceImpl implements UserDetailsService {

    @Autowired
    private SysUserMapper userMapper;
    @Autowired
    private SysRoleMapper roleMapper;
    @Autowired
    private SysMenuMapper menuMapper;

    @Override
    public UserDetails loadUserByUsername(String username) throws UsernameNotFoundException {
        // 1. 查询用户
        SysUser user = userMapper.selectByUsername(username);
        if (user == null) {
            throw new UsernameNotFoundException("用户不存在：" + username);
        }
        if (user.getStatus() != 1) {
            throw new DisabledException("用户已被禁用");
        }

        // 2. 查询用户角色
        List<SysRole> roles = roleMapper.selectRolesByUserId(user.getId());
        Set<String> roleKeys = roles.stream()
                .map(SysRole::getRoleKey)
                .collect(Collectors.toSet());

        // 3. 查询用户权限标识
        Set<String> permissions;
        if (roleKeys.contains("admin")) {
            // 超级管理员拥有所有权限
            permissions = Set.of("*:*:*");
        } else {
            List<SysMenu> menus = menuMapper.selectMenusByUserId(user.getId());
            permissions = menus.stream()
                    .map(SysMenu::getPermission)
                    .filter(StringUtils::hasText)
                    .collect(Collectors.toSet());
        }

        // 4. 查询用户数据权限范围
        Integer dataScope = roles.stream()
                .map(SysRole::getDataScope)
                .max(Integer::compareTo)
                .orElse(1);

        // 5. 构建登录用户对象
        LoginUser loginUser = new LoginUser();
        loginUser.setUserId(user.getId());
        loginUser.setUsername(user.getUsername());
        loginUser.setDeptId(user.getDeptId());
        loginUser.setRoles(roleKeys);
        loginUser.setPermissions(permissions);
        loginUser.setDataScope(dataScope);

        return loginUser;
    }
}
```

**权限校验AOP：**

```java
@Aspect
@Component
public class PermissionAspect {

    @Before("@annotation(requiresPermission)")
    public void checkPermission(RequiresPermission requiresPermission) {
        LoginUser loginUser = getLoginUser();
        if (loginUser == null) {
            throw new BusinessException(401, "未登录");
        }

        Set<String> permissions = loginUser.getPermissions();

        // 超级管理员
        if (permissions.contains("*:*:*")) {
            return;
        }

        String[] requiredPerms = requiresPermission.value();
        Logical logical = requiresPermission.logical();

        if (logical == Logical.AND) {
            for (String perm : requiredPerms) {
                if (!permissions.contains(perm)) {
                    throw new BusinessException(403, "没有权限：" + perm);
                }
            }
        } else {
            boolean hasAny = false;
            for (String perm : requiredPerms) {
                if (permissions.contains(perm)) {
                    hasAny = true;
                    break;
                }
            }
            if (!hasAny) {
                throw new BusinessException(403, "没有所需权限");
            }
        }
    }

    private LoginUser getLoginUser() {
        Authentication authentication = SecurityContextHolder.getContext()
                .getAuthentication();
        if (authentication != null && authentication.getPrincipal() instanceof LoginUser) {
            return (LoginUser) authentication.getPrincipal();
        }
        return null;
    }
}
```

```java
        return invocation.proceed();

    private String buildSqlCondition(DataScope dataScope, LoginUser loginUser) {
        String deptAlias = dataScope.deptAlias();
        String userAlias = dataScope.userAlias();

        switch (loginUser.getDataScope()) {
            case 1: // 全部数据
                return "";
            case 2: // 本部门数据

                    deptAlias, deptAlias);
            case 4: // 仅本人数据
                return String.format(" AND %s.id = %d",
                    userAlias, loginUser.getUserId());
            default:
                return "";
        }
```

```java
            logEntity.setUsername(loginUser.getUsername());

        try {
            Object result = point.proceed();
            logEntity.setStatus(1); // 成功
            logEntity.setDuration(System.currentTimeMillis() - startTime);
            return result;
        } catch (Exception e) {
            logEntity.setStatus(0); // 失败
            logEntity.setErrorMsg(e.getMessage());
            logEntity.setDuration(System.currentTimeMillis() - startTime);
            throw e;
        } finally {
            // 异步保存日志
```

```java
    @Autowired
    private JwtUtil jwtUtil;
    @Autowired
    private StringRedisTemplate redisTemplate;

    @PostMapping("/login")
    @Operation(summary = "用户登录")
    public Result<LoginVO> login(@RequestBody @Valid LoginDTO dto) {
        // 1. 认证
        Authentication authentication = authenticationManager.authenticate(
            new UsernamePasswordAuthenticationToken(dto.getUsername(), dto.getPassword()));

        LoginUser loginUser = (LoginUser) authentication.getPrincipal();

        // 2. 生成Token
        String token = jwtUtil.generateToken(loginUser.getUsername());

        // 3. 将用户信息存入Redis（用于Token校验和权限缓存）
        String redisKey = "login:token:" + token;
        redisTemplate.opsForValue().set(redisKey,
            JSON.toJSONString(loginUser), 24, TimeUnit.HOURS);

        // 4. 构建返回
        LoginVO loginVO = new LoginVO();
        loginVO.setToken(token);
        loginVO.setUserId(loginUser.getUserId());
        loginVO.setUsername(loginUser.getUsername());
        loginVO.setRoles(loginUser.getRoles());
        loginVO.setPermissions(loginUser.getPermissions());

        log.info("用户登录成功：{}", dto.getUsername());
        return Result.success(loginVO);
    }

    @PostMapping("/logout")
    @Operation(summary = "退出登录")
    public Result<Void> logout(HttpServletRequest request) {
        String token = request.getHeader("Authorization");
        if (token != null && token.startsWith("Bearer ")) {
            token = token.substring(7);
            redisTemplate.delete("login:token:" + token);
        }
        return Result.success();
    }

    @GetMapping("/info")
    @Operation(summary = "获取当前用户信息")
    public Result<UserInfoVO> getUserInfo() {
        LoginUser loginUser = getLoginUser();
        UserInfoVO vo = new UserInfoVO();
        vo.setUserId(loginUser.getUserId());
        vo.setUsername(loginUser.getUsername());
        vo.setRoles(loginUser.getRoles());
        vo.setPermissions(loginUser.getPermissions());
        return Result.success(vo);
    }

    private LoginUser getLoginUser() {
        return (LoginUser) SecurityContextHolder.getContext()
                .getAuthentication().getPrincipal();
    }
}
```

## 7.5 学习建议

### 项目选择建议

### 学习方法建议

1. **先模仿后创新**：参考开源项目，理解设计思路后再加入自己的改进
2. **小步快跑**：将大项目拆分为小模块，逐个实现
3. **持续重构**：完成功能后回顾代码，不断优化
4. **写文档**：记录设计思路、遇到的问题及解决方案
5. **代码审查**：找同学或社区帮忙 Review 代码

### 常见问题解答

**Q1：项目做到什么程度才算完成？**

A：满足以下条件即可：
- 核心功能完整可用
- 代码规范，有适当注释
- 有基本的单元测试
- 有完整的 API 文档
- 能够部署运行

**Q2：遇到不会的技术怎么办？**

A：
1. 先查阅官方文档
2. 搜索相关教程和博客
3. 参考开源项目的实现
4. 在社区提问（StackOverflow、掘金等）
5. 适当降低要求，先用简单方案实现

**Q3：如何避免项目烂尾？**

A：
1. 控制项目规模，不要贪大求全
2. 制定明确的里程碑和截止时间
3. 优先完成核心功能
4. 记录进度，保持成就感
5. 找伙伴一起学习和监督
